let allData = {}; let currentMatches = []; async function loadMatches() { try { const selectedDate = document.getElementById('match-date').value; // Eğer veri henüz çekilmediyse data.json'ı oku if (Object.keys(allData).length === 0) { const res = await fetch('data.json'); allData = await res.json(); } // Seçilen tarihe ait maçları çek currentMatches = allData[selectedDate] || []; const grid = document.getElementById('match-grid'); grid.innerHTML = ''; if (!currentMatches || currentMatches.length === 0) { grid.innerHTML = `
Seçilen tarih (${selectedDate}) için maç bulunamadı.
`; return; } currentMatches.forEach((m, idx) => { grid.innerHTML += `
${m.league}
${m.home}${m.analysis.exp_home}
${m.away}${m.analysis.exp_away}
MAÇ TOPLAMI
${m.analysis.match_total}
İLK YARI
${m.analysis.first_half}
`; }); } catch (e) { console.error("Veri okunamadı:", e); } }
